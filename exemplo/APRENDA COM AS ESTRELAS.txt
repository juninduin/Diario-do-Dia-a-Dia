<script>
const EMOTIONS = [
  {id:'feliz', label:'Feliz', emoji:'😄'},
  {id:'triste', label:'Triste', emoji:'😢'},
  {id:'bravo', label:'Bravo(a)', emoji:'😠'},
  {id:'medo', label:'Com medo', emoji:'😨'},
  {id:'surpreso', label:'Surpreso(a)', emoji:'😮'},
  {id:'calmo', label:'Calmo(a)', emoji:'😌'},
  {id:'vergonha', label:'Envergonhado(a)', emoji:'😳'},
  {id:'animado', label:'Animado(a)', emoji:'🤩'},
];

const REACTIONS = [
  {emoji:'😭', label:'Chorei'},
  {emoji:'📣', label:'Gritei'},
  {emoji:'🙈', label:'Me escondi'},
  {emoji:'🗣️', label:'Conversei sobre isso'},
  {emoji:'😆', label:'Ri bastante'},
  {emoji:'🤗', label:'Abracei alguém'},
  {emoji:'🤐', label:'Fiquei quieto(a)'},
  {emoji:'👊', label:'Bati o pé / empurrei'},
];

const SOLUTIONS = [
  {emoji:'🌬️', label:'Respirar fundo 3 vezes'},
  {emoji:'🙋', label:'Pedir ajuda a um adulto'},
  {emoji:'💬', label:'Falar o que eu senti'},
  {emoji:'🔢', label:'Contar até 10'},
  {emoji:'🤗', label:'Pedir um abraço'},
  {emoji:'🎨', label:'Desenhar o que senti'},
];

let state = { emotion:null, reaction:null, reactionCustom:'', solution:null, solutionCustom:'' };
let step = 0; // 0 emotion,1 reaction,2 solution,3 summary

const mascotEl = document.getElementById('mascot');
const subheader = document.getElementById('subheader');
const cardArea = document.getElementById('cardArea');
const progress = document.getElementById('progress');
const diaryLink = document.getElementById('diaryLink');

function setMascot(moodId){
  mascotEl.className = 'mascot' + (moodId ? ' ' + moodId : '');
}

function updateProgress(){
  [...progress.children].forEach((d,i)=>d.classList.toggle('active', i===step));
  progress.style.display = step<=3 ? 'flex':'none';
}

function renderEmotionStep(){
  subheader.textContent = 'Como você se sentiu hoje?';
  setMascot(state.emotion || null);
  cardArea.innerHTML = `
    <h2 class="step-title">Escolha a carinha de hoje 🌈</h2>
    <p class="step-sub">Toque na emoção que mais combina com o seu dia</p>
    <div class="grid" id="emoGrid"></div>
    <div class="nav">
      <span></span>
      <button class="btn btn-primary" id="nextBtn" disabled>Próximo ➜</button>
    </div>
  `;
  const grid = document.getElementById('emoGrid');
  EMOTIONS.forEach(e=>{
    const div = document.createElement('div');
    div.className = 'opt' + (state.emotion===e.id ? ' selected':'');
    div.innerHTML = `<span class="emoji">${e.emoji}</span><span>${e.label}</span>`;
    div.onclick = ()=>{
      state.emotion = e.id;
      setMascot(e.id);
      renderEmotionStep();
    };
    grid.appendChild(div);
  });
  document.getElementById('nextBtn').disabled = !state.emotion;
  document.getElementById('nextBtn').onclick = ()=>{ step=1; render(); };
}

function renderReactionStep(){
  const emo = EMOTIONS.find(e=>e.id===state.emotion);
  subheader.textContent = `Você se sentiu ${emo.label.toLowerCase()} ${emo.emoji}`;
  cardArea.innerHTML = `
    <h2 class="step-title">O que você fez?</h2>
    <p class="step-sub">Escolha o que mais combina, ou escreva o seu jeito</p>
    <div class="grid single-col" id="reactGrid"></div>
    <textarea id="reactCustom" rows="2" placeholder="Ou escreva com suas palavras...">${state.reactionCustom||''}</textarea>
    <div class="nav">
      <button class="btn btn-ghost" id="backBtn">⟲ Voltar</button>
      <button class="btn btn-primary" id="nextBtn">Próximo ➜</button>
    </div>
  `;
  const grid = document.getElementById('reactGrid');
  REACTIONS.forEach(r=>{
    const div = document.createElement('div');
    div.className = 'opt wide' + (state.reaction===r.label ? ' selected':'');
    div.innerHTML = `<span class="emoji">${r.emoji}</span><span>${r.label}</span>`;
    div.onclick = ()=>{
      state.reaction = r.label;
      document.getElementById('reactCustom').value='';
      state.reactionCustom='';
      renderReactionStep();
    };
    grid.appendChild(div);
  });
  document.getElementById('reactCustom').oninput = (e)=>{
    state.reactionCustom = e.target.value;
    if(e.target.value.trim()) state.reaction = null;
  };
  document.getElementById('backBtn').onclick = ()=>{ step=0; render(); };
  document.getElementById('nextBtn').onclick = ()=>{ step=2; render(); };
}

function renderSolutionStep(){
  cardArea.innerHTML = `
    <h2 class="step-title">Da próxima vez...</h2>
    <p class="step-sub">O que você poderia fazer para resolver melhor?</p>
    <div class="grid single-col" id="solGrid"></div>
    <textarea id="solCustom" rows="2" placeholder="Ou escreva sua própria ideia...">${state.solutionCustom||''}</textarea>
    <div class="nav">
      <button class="btn btn-ghost" id="backBtn">⟲ Voltar</button>
      <button class="btn btn-primary" id="nextBtn">Ver resumo 🎉</button>
    </div>
  `;
  const grid = document.getElementById('solGrid');
  SOLUTIONS.forEach(s=>{
    const div = document.createElement('div');
    div.className = 'opt wide' + (state.solution===s.label ? ' selected':'');
    div.innerHTML = `<span class="emoji">${s.emoji}</span><span>${s.label}</span>`;
    div.onclick = ()=>{
      state.solution = s.label;
      document.getElementById('solCustom').value='';
      state.solutionCustom='';
      renderSolutionStep();
    };
    grid.appendChild(div);
  });
  document.getElementById('solCustom').oninput = (e)=>{
    state.solutionCustom = e.target.value;
    if(e.target.value.trim()) state.solution = null;
  };
  document.getElementById('backBtn').onclick = ()=>{ step=1; render(); };
  document.getElementById('nextBtn').onclick = async ()=>{ await saveEntry(); step=3; render(); };
}

async function saveEntry(){
  const today = new Date();
  const dateKey = today.toISOString().slice(0,10);
  const emo = EMOTIONS.find(e=>e.id===state.emotion);
  const entry = {
    date: dateKey,
    displayDate: today.toLocaleDateString('pt-BR', {weekday:'long', day:'2-digit', month:'long'}),
    emotionEmoji: emo.emoji,
    emotionLabel: emo.label,
    reaction: state.reaction || state.reactionCustom || '—',
    solution: state.solution || state.solutionCustom || '—',
  };
  try{
    await window.storage.set('entry:' + dateKey + ':' + Date.now(), JSON.stringify(entry), false);
  }catch(err){
    console.error('Erro ao salvar', err);
  }
}

function renderSummaryStep(){
  const emo = EMOTIONS.find(e=>e.id===state.emotion);
  subheader.textContent = 'Muito bem! 🎉';
  setMascot(state.emotion);
  cardArea.innerHTML = `
    <h2 class="step-title">Resumo do seu dia</h2>
    <div class="summary-line"><span class="label">Senti</span><span class="val">${emo.emoji} ${emo.label}</span></div>
    <div class="summary-line"><span class="label">Eu fiz</span><span class="val">${state.reaction || state.reactionCustom}</span></div>
    <div class="summary-line"><span class="label">Da próxima vez</span><span class="val">${state.solution || state.solutionCustom}</span></div>
    <p class="praise">Cada emoção é bem-vinda. Aprender a lidar com elas é uma superpotência! 💪</p>
    <div class="nav">
      <button class="btn btn-ghost" id="diaryBtn">📖 Ver diário</button>
      <button class="btn btn-primary" id="restartBtn">Jogar de novo 🔄</button>
    </div>
  `;
  document.getElementById('restartBtn').onclick = ()=>{
    state = { emotion:null, reaction:null, reactionCustom:'', solution:null, solutionCustom:'' };
    step = 0;
    render();
  };
  document.getElementById('diaryBtn').onclick = renderDiary;
}

async function renderDiary(){
  subheader.textContent = 'Seu diário das emoções';
  setMascot(null);
  progress.style.display = 'none';
  cardArea.innerHTML = `<h2 class="step-title">📖 Meu diário</h2><div id="diaryList"><p class="empty">Carregando...</p></div>
    <div class="nav"><button class="btn btn-primary" id="backToGame">Voltar ao joguinho</button></div>`;
  document.getElementById('backToGame').onclick = ()=>{
    state = { emotion:null, reaction:null, reactionCustom:'', solution:null, solutionCustom:'' };
    step = 0;
    render();
  };
  const listEl = document.getElementById('diaryList');
  try{
    const res = await window.storage.list('entry:', false);
    const keys = (res && res.keys) ? res.keys : [];
    if(keys.length===0){
      listEl.innerHTML = '<p class="empty">Ainda não há registros. Jogue para começar seu diário! 🌟</p>';
      return;
    }
    const entries = [];
    for(const k of keys){
      try{
        const r = await window.storage.get(k, false);
        if(r && r.value) entries.push(JSON.parse(r.value));
      }catch(e){}
    }
    entries.sort((a,b)=> b.date.localeCompare(a.date));
    listEl.innerHTML = entries.map(en => `
      <div class="diary-entry">
        <div class="date">${en.emotionEmoji} ${capitalize(en.displayDate)}</div>
        <div class="row"><b>Senti:</b> ${en.emotionLabel}</div>
        <div class="row"><b>Fiz:</b> ${en.reaction}</div>
        <div class="row"><b>Próxima vez:</b> ${en.solution}</div>
      </div>
    `).join('');
  }catch(err){
    listEl.innerHTML = '<p class="empty">Não consegui carregar o diário agora.</p>';
  }
}

function capitalize(s){ return s.charAt(0).toUpperCase() + s.slice(1); }

function render(){
  updateProgress();
  if(step===0) renderEmotionStep();
  else if(step===1) renderReactionStep();
  else if(step===2) renderSolutionStep();
  else if(step===3) renderSummaryStep();
}

diaryLink.onclick = renderDiary;

render();
</script>
